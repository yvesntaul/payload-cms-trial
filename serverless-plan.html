# 2. Choose the appropriate template for your Payload version

Payload Configuration:

// payload.config.ts
import { postgresAdapter } from '@payloadcms/db-postgres'
import { s3Storage } from '@payloadcms/storage-s3'
import { buildConfig } from 'payload'

export default buildConfig({
  // Use standard PostgreSQL adapter with Neon connection string
  db: postgresAdapter({
    pool: {
      connectionString: process.env.DATABASE_URL, // Neon CMS connection string
    },
  }),
  
  // Use S3-compatible adapter for R2
  plugins: [
    s3Storage({
      collections: {
        media: true,
      },
      config: {
        endpoint: process.env.R2_ENDPOINT,
        region: 'auto',
        credentials: {
          accessKeyId: process.env.R2_ACCESS_KEY_ID,
          secretAccessKey: process.env.R2_SECRET_ACCESS_KEY,
        },
      },
      bucket: process.env.R2_BUCKET,
    }),
  ],
})

# 1. Create Next.js app
npx create-next-app@latest public-website --typescript

# 2. Fetch content from Payload API using REST
// lib/payload-client.ts
export async function getPublishedArtifacts() {
  const response = await fetch(
    `${process.env.NEXT_PUBLIC_PAYLOAD_URL}/artifacts?where[_status][equals]=published`,
    {
      headers: {
        'Content-Type': 'application/json',
      },
      // ✅ Add revalidation for caching
      next: { revalidate: 60 }, // Revalidate every 60 seconds
    }
  )
  
  if (!response.ok) {
    throw new Error(`Failed to fetch artifacts: ${response.status}`)
  }
  
  const data = await response.json()
  return data.docs
}

Environment Variables:

NEXT_PUBLIC_PAYLOAD_URL=https://your-payload.vercel.app/api
NEXT_PUBLIC_APP_URL=https://your-app.onrender.com

Cache/Revalidation Policy: The public website should implement incremental static regeneration (ISR) or revalidation to ensure content updates are reflected without manual rebuilding.

Critical Design Rule: The public website should never import Payload configuration directly or access the database directly. It must use HTTP REST calls to the Payload API.

3. Critical Architecture Decisions

3.1 Assessment Security

Problem: The artifact model contains is_correct, but the public API must not return correct answers.

Solution: Hide is_correct at the Payload field access level, AND protect the Express endpoint.

Payload Field Access Control:

// collections/Artifacts.ts
{
  name: 'assessment_options',
  type: 'array',
  fields: [
    {
      name: 'text',
      type: 'text',
    },
    {
      name: 'is_correct',
      type: 'checkbox',
      // ✅ Hide from public API
      access: {
        read: ({ req }) => {
          // Only allow admin/editor to see correct answers
          return req.user?.role === 'admin' || req.user?.role === 'editor'
        },
      },
    },
  ],
}

Express Assessment Endpoint (Protected):

// Express service
export class AssessmentService {
  // ✅ Use server-to-server auth, NOT client-facing
  async getAssessment(artifactId: string) {
    const response = await fetch(
      `${process.env.PAYLOAD_API_URL}/artifacts/${artifactId}`,
      {
        headers: {
          // Server-to-server authentication (not exposed to Unity)
          'x-service-token': process.env.SERVICE_TOKEN,
        },
      }
    )
    const artifact = await response.json()
    
    // Return only questions without is_correct
    return {
      questions: artifact.assessment_options.map((opt: any) => ({
        text: opt.text,
        // is_correct is removed by Payload access control
      })),
    }
  }
  
  async submitAssessment(attempt: AssessmentAttempt) {
    // Score internally
    // Never expose is_correct to Unity client
    const score = this.calculateScore(attempt)
    return { score, total: attempt.totalQuestions }
  }
}

3.2 Service Token Authentication (Server-Side Only)

🚨 CRITICAL: Service tokens must NEVER be sent from browser-side code.

Correct (Server Component in Payload):

// In Payload server component or custom endpoint
// ✅ Runs on server, token is safe
export const getAnalyticsData = async () => {
  const response = await fetch(
    'https://your-api.onrender.com/api/analytics/overview',
    {
      headers: {
        'x-service-token': process.env.SERVICE_TOKEN, // ✅ Safe on server
      },
    }
  )
  return response.json()
}

❌ INCORRECT (Browser-side component):

// In Payload admin component (client-side)
// ❌ NEVER DO THIS - token would be exposed to browser
const fetchAnalytics = async () => {
  const response = await fetch(
    'https://your-api.onrender.com/api/analytics/overview',
    {
      headers: {
        'x-service-token': process.env.SERVICE_TOKEN, // ❌ EXPOSED!
      },
    }
  )
}

3.3 Reports Access (Cross-Project)

Problem: With two separate Neon projects, Payload cannot directly query Express reports.

Solution: Express API is the authoritative owner of reports.

Payload Admin Component (Server Component)
         ↓
Authenticated HTTP Request (with SERVICE_TOKEN)
         ↓
Express API (GET /api/analytics/reports)
         ↓
Neon App Database
         ↓
Returns reports to Payload Admin

Express endpoint:

app.get('/api/analytics/reports', verifyServiceToken, async (req, res) => {
  const reports = await db.query.reports.findMany()
  res.json(reports)
})

Key: Reports are NOT stored in Payload collections. Express is the authoritative owner.

6. Backup and Recovery

6.1 Backup Strategy

Critical: Platform backups are not sufficient. Implement the following:

Requirement Implementation
Export frequency Daily automated export to off-provider location
Retention period 30 days minimum
Off-provider backup Separate cloud storage (e.g., another provider, not just R2)
Restore testing Quarterly restore test
R2 asset recovery Enable versioning (verify availability)
Account recovery Document recovery procedures for each platform

6.2 Backup Script

# backup.sh
#!/bin/bash
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_DIR="./backups"

# Export Neon CMS database
PGSSLMODE=require pg_dump $NEON_CMS_URL > "$BACKUP_DIR/cms_$DATE.sql"

# Export Neon App database
PGSSLMODE=require pg_dump $NEON_APP_URL > "$BACKUP_DIR/app_$DATE.sql"

# Compress
gzip "$BACKUP_DIR/cms_$DATE.sql"
gzip "$BACKUP_DIR/app_$DATE.sql"

# Upload to off-provider backup location
# (Not Cloudflare R2 - use a different provider)
rclone copy "$BACKUP_DIR/" remote:backup-bucket/ --include "*.sql.gz"

# Keep last 30 days locally
find "$BACKUP_DIR" -name "*.sql.gz" -mtime +30 -delete

# Keep last 30 days on remote (already handled by rclone sync)

6.3 Restore Commands

# Restore CMS database
PGSSLMODE=require psql $NEON_CMS_URL < backups/cms_20260829_120000.sql

# Restore App database
PGSSLMODE=require psql $NEON_APP_URL < backups/app_20260829_120000.sql

6.4 R2 Asset Recovery

# If using R2 versioning, list versions
aws s3api list-object-versions --bucket artifacts-media --endpoint-url $R2_ENDPOINT

# Restore specific version
aws s3api restore-object --bucket artifacts-media --key models/artifact-001.glb --version-id <version-id>

Verify R2 versioning: Confirm that R2 versioning is available and enabled for your account. Document the exact recovery process.
